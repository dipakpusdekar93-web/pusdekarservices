function renderStatementTable(dataList) { const table = document.getElementById("statementTableEntries"); table.innerHTML = ""; const selectedWallet = document.getElementById('walletFilter').value; let runningBal = 0; const axisCustomName = localStorage.getItem("wallet_axis") || "Axis Bank"; const sbiCustomName = localStorage.getItem("wallet_sbi") || "SBI"; const boiCustomName = localStorage.getItem("wallet_boi") || "Bank of India"; const unionCustomName = localStorage.getItem("wallet_union") || "Union Bank"; dataList.forEach((data) => { let dateStr = "Loading..."; if (data.timestamp) { dateStr = data.timestamp.toDate().toLocaleString('en-IN'); } const amt = parseFloat(data.amount) || 0; const comm = parseFloat(data.commission) || 0; const bSrc = data.bankSource || "Axis"; const eSrc = data.expenseSource || "Cash"; // RULE 2: In "All Statement", Grand Total/Running Balance grows by Commission (+) and reduces by Shop Expenses (-) if (selectedWallet === "All") { if(data.service.includes("Opening")) { runningBal += amt; } else if(data.service === "Shop Expense" || data.service === "Fino Cash Deposit") { runningBal -= amt; } if(comm > 0) { runningBal += comm; } } else { // RULE 1: In Specific Wallet Statement, show actual money flow (plus/minus) and remaining balance of that wallet let impactAmt = 0; if (selectedWallet === "Fino") { if (data.service === "Fino Opening") impactAmt = amt; else if (data.service === "Fino Payment Bank") impactAmt = amt; else if (data.service === "Fino Cash Deposit") impactAmt = -amt; else if (data.service === "Shop Expense" && eSrc === "Fino") impactAmt = -amt; if(comm > 0) impactAmt += comm; // Commission adds to wallet profit } else if (selectedWallet === "Payworld") { if (data.service === "Payworld Opening") impactAmt = amt; else if (data.service === "Payworld Services") impactAmt = amt; else if (data.service === "Shop Expense" && eSrc === "Payworld") impactAmt = -amt; if(comm > 0) impactAmt += comm; } else if (selectedWallet === "Mahapower") { if (data.service === "Mahapower Opening") impactAmt = amt; else if (data.service === "Mahapower Bill Collection") impactAmt = -amt; else if (data.service === "Shop Expense" && eSrc === "Mahapower") impactAmt = -amt; } else if (selectedWallet === "Cash") { if (data.service === "Cash Opening") impactAmt = amt; else if (data.service === "Fino Payment Bank") impactAmt = -amt; else if (data.service === "Payworld Services") impactAmt = -amt; else if (data.service === "Mahapower Bill Collection") impactAmt = amt; else if (data.service === "Fino Cash Deposit") impactAmt = amt; else if (data.service === "Money Transfer") impactAmt = amt; else if (data.service === "UPI Cash Given") impactAmt = -amt; else if (data.service === "Printing & Xerox") impactAmt = amt; else if (data.service === "Shop Expense" && eSrc === "Cash") impactAmt = -amt; if(comm > 0) impactAmt += comm; } else { let bankKey = selectedWallet; if (data.service === `${bankKey} Opening`) impactAmt = amt; else if (data.service === "Money Transfer" && bSrc === bankKey) impactAmt = -amt; else if (data.service === "UPI Cash Given" && bSrc === bankKey) impactAmt = amt; else if (data.service === "Shop Expense" && eSrc === bankKey) impactAmt = -amt; } runningBal += impactAmt; } let serviceDisplay = data.service; let amtDisplay = `₹ ${data.amount}`; if(data.service === "Money Transfer" || data.service === "UPI Cash Given") { let dispBank = bSrc; if(bSrc === "Axis") dispBank = axisCustomName; else if(bSrc === "SBI") dispBank = sbiCustomName; else if(bSrc === "BOI") dispBank = boiCustomName; else if(bSrc === "Union") dispBank = unionCustomName; serviceDisplay += ` (${dispBank})`; } else if(data.service === "Shop Expense") { amtDisplay = `- ₹ ${data.amount} (${eSrc})`; } else if(data.service === "Fino Cash Deposit" || data.service === "Mahapower Bill Collection") { amtDisplay = `- ₹ ${data.amount} (${data.service.includes('Mahapower') ? 'Mahapower' : 'Fino'})`; } else if(!data.service.includes("Opening")) { amtDisplay = `+ ₹ ${data.amount}`; } let purposeText = data.purpose || "-"; if(data.narration && data.narration !== "-") { purposeText += `
📝 ${data.narration}`; } const row = ` ${dateStr} ${serviceDisplay} ${data.details}
${data.mobile} ${data.payStatus} ${purposeText} ${amtDisplay} + ₹ ${data.commission || 0} ₹ ${runningBal} `; table.innerHTML += row; }); } function filterData() { const selectedWallet = document.getElementById('walletFilter').value; const f = document.getElementById('fromDate').value; const t = document.getElementById('toDate').value; let filtered = allTransactions.filter(item => { let matchWallet = true; if(selectedWallet !== "All") { if(selectedWallet === "Fino") { matchWallet = (item.service.includes("Fino") || item.expenseSource === "Fino"); } else if(selectedWallet === "Payworld") { matchWallet = (item.service.includes("Payworld") || item.expenseSource === "Payworld"); } else if(selectedWallet === "Axis") { matchWallet = (item.service.includes("Axis") || item.bankSource === "Axis" || item.expenseSource === "Axis"); } else if(selectedWallet === "SBI") { matchWallet = (item.service.includes("SBI") || item.bankSource === "SBI" || item.expenseSource === "SBI"); } else if(selectedWallet === "BOI") { matchWallet = (item.service.includes("BOI") || item.bankSource === "BOI" || item.expenseSource === "BOI"); } else if(selectedWallet === "Union") { matchWallet = (item.service.includes("Union") || item.bankSource === "Union" || item.expenseSource === "Union"); } else if(selectedWallet === "Mahapower") { matchWallet = (item.service.includes("Mahapower") || item.expenseSource === "Mahapower"); } else if(selectedWallet === "Cash") { matchWallet = (item.service.includes("Cash") || item.service === "Fino Payment Bank" || item.service === "Payworld Services" || item.service === "Mahapower Bill Collection" || item.service === "Money Transfer" || item.service === "UPI Cash Given" || item.service === "Printing & Xerox" || item.expenseSource === "Cash"); } } return matchWallet; }); if(f && t) { const fromDate = new Date(f); fromDate.setHours(0,0,0,0); const toDate = new Date(t); toDate.setHours(23,59,59,999); filtered = filtered.filter(item => { if(!item.timestamp) return false; const d = item.timestamp.toDate(); return d >= fromDate && d <= toDate; }); } renderStatementTable(filtered); } function resetFilter() { document.getElementById('walletFilter').value = "All"; document.getElementById('fromDate').value = ""; document.getElementById('toDate').value = ""; renderStatementTable(allTransactions); }
